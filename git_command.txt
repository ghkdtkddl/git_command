git config --global user.name [name] //유저정보 등록
git config --unset --global user.name //유저정보 제거
git conifg --global --list //등록정보 확인

git config --global user.email [email] //유저 이메일 정보 등록
------------------------------------------------------------------
위 유저 정보는 C://Users/<windows 계정명>/.gitconfig 파일에 저장되어 있음.

------------------------------------------------------------------
git  config --global init.defaultBranch main // github의 기본 branch가 기존 master->main으로 변경되었으므로 main으로 해줌
git config --get init.defaultBranch // 기본 branch 설정 확인
git init // git project로 관리하기 위한 .git을 생성해줌.
git status //git 프로젝트의 파일 상태 확인
git add . //git commit할 파일을 추가해줌, add뒤에 .일 경우 모든 파일 선택
git commit -m '명령어' // git commit 명렁어
------------------------------------------------------------------
git remote add origin [git repository]
git push -u origin main // push

------------------------------------------------------------------
push 실행 순서.

1. 옮기고자 하는 파일 오른쪽 마우스 클릭 - git bash here
2. git config --global user.name [name] //유저정보 등록
3. git config --global user.email [email] //유저정보 등록
4, git init // git 프로젝트 관리를 위함
5. git add [경로/파일명] or git add .// 선택적 파일 혹은 모든 파일 추가
6. git status // git 프로젝트의 파일 상태 확인
7. git commit -m '명령어' // git commit 명령어 작성
8. git branch -M main // 기본 브랜치를 master에서 main으로 변경 #무슨차이인지 확인해보기
9. git remote add origin [git repository url] // github repository 지정
10. git push -u origin main // 지정한 repository에 add한 파일을 push하기

------------------------------------------------------------------
pull실행 순서

1. git init // git 프로젝트 관리를 위함
2. git pull [git repository] // git repository에 있는 파일 불러오기

------------------------------------------------------------------
에러 정리

1. git push -u origin main 명령 시, 
" ! [rejected]        main -> main (fetch first)
error: failed to push some refs to 'https://github.com/ghkdtkddl/git_command.git'"
에러 발생 했을 때, 최신 커밋을 반영하기 위해 'git pull origin main --rebase' 입력하여 커밋이 병합됨.
그 후, git push origin main을 입력하면 push가 됨
=> push 전에는 항상 pull을 입력하기!
